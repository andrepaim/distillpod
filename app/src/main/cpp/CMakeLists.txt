cmake_minimum_required(VERSION 3.22.1)
project(distillpod_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
# Native inference must be optimized even when the Android UI is debuggable.
# Keep symbols for native crash reports; -O0 makes the CPU model impractical.
set(CMAKE_C_FLAGS_DEBUG "-O2 -g")
set(CMAKE_CXX_FLAGS_DEBUG "-O2 -g")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
add_subdirectory(../../../../third_party/whisper.cpp whisper)
add_library(distillpod_bridge STATIC WhisperBridge.cpp)
target_link_libraries(distillpod_bridge PUBLIC whisper)
target_include_directories(distillpod_bridge PUBLIC .)
if(ANDROID)
    add_library(distillpod_whisper SHARED JniBridge.cpp)
    target_link_libraries(distillpod_whisper PRIVATE distillpod_bridge log)
    # NDK r27 needs explicit ELF LOAD and RELRO alignment for 16 KB Android devices.
    target_link_options(distillpod_whisper PRIVATE
        "-Wl,-z,max-page-size=16384"
        "-Wl,-z,common-page-size=16384")
else()
    enable_testing()
    add_executable(transcriber_test ../../../../native-tests/transcriber_test.cpp)
    target_link_libraries(transcriber_test PRIVATE distillpod_bridge)
    add_test(NAME native_input_validation COMMAND transcriber_test)
endif()
