cmake_minimum_required(VERSION 3.22.1)
project(distillpod_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
add_subdirectory(../../../../third_party/whisper.cpp whisper)
add_library(distillpod_bridge STATIC WhisperBridge.cpp)
target_link_libraries(distillpod_bridge PUBLIC whisper)
target_include_directories(distillpod_bridge PUBLIC .)
if(ANDROID)
    add_library(distillpod_whisper SHARED JniBridge.cpp)
    target_link_libraries(distillpod_whisper PRIVATE distillpod_bridge log)
else()
    enable_testing()
    add_executable(transcriber_test ../../../../native-tests/transcriber_test.cpp)
    target_link_libraries(transcriber_test PRIVATE distillpod_bridge)
    add_test(NAME native_input_validation COMMAND transcriber_test)
endif()
